
# 1. Import Libraries

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import seaborn as sns


# 2. Load Dataset

df = pd.read_csv("uber.csv")

print(df.head())


# 3. Check Dataset Information

print(df.info())
print("Dataset Shape:", df.shape)


# 4. Preprocessing

# Check missing values
print(df.isnull().sum())

# Remove missing values
df.dropna(inplace=True)

# Remove unwanted columns
df.drop(['Unnamed: 0', 'key'], axis=1, inplace=True)

print(df.head())


# 5. Identify Outliers

def find_outliers_IQR(data):
    q1 = data.quantile(0.25)
    q3 = data.quantile(0.75)
    IQR = q3 - q1

    outliers = data[
        (data < q1 - 1.5 * IQR) |
        (data > q3 + 1.5 * IQR)
    ]

    return outliers

outliers = find_outliers_IQR(df['fare_amount'])

print("Number of Outliers:", len(outliers))
print("Maximum Outlier:", outliers.max())
print("Minimum Outlier:", outliers.min())


# 6. Visualize Data

sns.histplot(df['fare_amount'], kde=True)
plt.show()


# 7. Check Correlation

corr = df.select_dtypes(include='number').corr()

sns.heatmap(corr, annot=True)
plt.show()


# 8. Convert Date and Time

df['pickup_datetime'] = pd.to_datetime(df['pickup_datetime'])

df['day'] = df['pickup_datetime'].dt.day
df['hour'] = df['pickup_datetime'].dt.hour
df['month'] = df['pickup_datetime'].dt.month
df['year'] = df['pickup_datetime'].dt.year
df['weekday'] = df['pickup_datetime'].dt.weekday

df.drop(['pickup_datetime'], axis=1, inplace=True)


# 9. Display Features and Target

x = df.drop('fare_amount', axis=1)
y = df['fare_amount']

print("Input Features:")
print(x.columns)

print("Target Variable:")
print(y.name)

print("Feature Data:")
print(x.head())


# 10. Split Dataset

from sklearn.model_selection import train_test_split

x_train, x_test, y_train, y_test = train_test_split(
    x, y, test_size=0.2, random_state=101
)

print("Training Data:", x_train.shape)
print("Testing Data:", x_test.shape)


# 11. Linear Regression

from sklearn.linear_model import LinearRegression

lrmodel = LinearRegression()

lrmodel.fit(x_train, y_train)

lr_pred = lrmodel.predict(x_test)


# 12. Random Forest Regression

from sklearn.ensemble import RandomForestRegressor

rfrmodel = RandomForestRegressor(
    n_estimators=100,
    random_state=101
)

rfrmodel.fit(x_train, y_train)

rfr_pred = rfrmodel.predict(x_test)


# 13. Evaluate Both Models

from sklearn.metrics import mean_squared_error, r2_score

lr_rmse = np.sqrt(mean_squared_error(y_test, lr_pred))
lr_r2 = r2_score(y_test, lr_pred)

rfr_rmse = np.sqrt(mean_squared_error(y_test, rfr_pred))
rfr_r2 = r2_score(y_test, rfr_pred)

print("Linear Regression RMSE:", lr_rmse)
print("Linear Regression R2:", lr_r2)

print("Random Forest RMSE:", rfr_rmse)
print("Random Forest R2:", rfr_r2)


# 14. Compare Both Models

result = pd.DataFrame({
    'Model': ['Linear Regression', 'Random Forest'],
    'RMSE': [lr_rmse, rfr_rmse],
    'R2 Score': [lr_r2, rfr_r2]
})

print(result)


# 15. Find Better Model

if rfr_rmse < lr_rmse and rfr_r2 > lr_r2:
    print("Random Forest is Better")
elif lr_rmse < rfr_rmse and lr_r2 > rfr_r2:
    print("Linear Regression is Better")
else:
    print("Compare both scores to decide the better model")


# 16. Predict Uber Fare

print("Actual Fare:", y_test.iloc[0])
print("Predicted Fare:", rfr_pred[0])
